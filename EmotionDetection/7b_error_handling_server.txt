from flask import Flask, render_template, request
from EmotionDetection.emotion_detection import emotion_detector

app = Flask("Emotion Detector")

@app.route("/")
def render_index_page():
    ''' Renders the main application page '''
    return render_template('index.html')

@app.route("/emotionDetector")
def detect_emotions():
    text_to_analyze = request.args.get('textToAnalyze', '')
    if not text_to_analyze:
        return "Invalid text! Please try again!."
    #if not text_to_analyze:
        #return "Error: Missing 'textToAnalyze' query parameter", 400

    try:
        response = emotion_detector(text_to_analyze)
        dominant_emotion = response.get('dominant_emotion')
        filtered_emotions = response.get('emotions')

        #If blank/invalid text was sent, emotion_detector returns None values
        if dominant_emotion is None:
            return "Invalid text! Please try again!."

        response_str = (
            f"For the given statement, the system response is "
            f"'anger': {filtered_emotions.get('anger', 0)}, "
            f"'disgust': {filtered_emotions.get('disgust', 0)}, "
            f"'fear': {filtered_emotions.get('fear', 0)}, "
            f"'joy': {filtered_emotions.get('joy', 0)} and "
            f"'sadness': {filtered_emotions.get('sadness', 0)}. "
            f"The dominant emotion is {dominant_emotion}."
        )
        return response_str

    except Exception as e:
        return f"Internal Server Error: {e}", 500

if __name__ == '__main__':
    app.run(host='localhost', port=5000)
